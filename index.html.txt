  // Плавное появление блоков при скролле
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        entry.target.classList.add('visible');
      }
    });
  }, { threshold: 0.15 });

  document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

  // Клик по диско-шару — воспроизведение припева

  const chorus = document.getElementById('chorusAudio');

  discoBall.addEventListener('click', () => {
    // Сброс и воспроизведение
    chorus.currentTime = 0;
    chorus.play().catch(err => {
      console.log('Не удалось воспроизвести аудио:', err);
    });

    // Визуальный отклик
    discoBall.style.transform = 'scale(0.9)';
    setTimeout(() => {
      discoBall.style.transform = '';
    }, 150);
  });

  // Если аудио закончилось — сбросить
  chorus.addEventListener('ended', () => {
    chorus.currentTime = 0;
